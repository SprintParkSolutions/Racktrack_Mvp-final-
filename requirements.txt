# Root requirements for the full RackTrack stack (CV pipeline + integrations).
# The Dockerfile installs these for the unified container; servicenow/ has its
# own narrower requirements.txt for the standalone ServiceNow bridge.
#
# These are loose (>=) floors for development. For reproducible builds/deploys
# use the pinned lockfile instead:  pip install -r requirements.lock.txt
# (see requirements.lock.txt for how it was generated and known caveats).
#
# WARNING — the Dockerfile and CI currently install THIS file, not the lock.
# Loose floors mean a fresh install pulls the newest torch/opencv wheels (built
# for numpy 2.x) while numpy is held at <2.0 below — pip resolves it, but the
# numpy-2.x-era wheels can abort at import against a numpy-1.26 runtime (numpy
# 2.0 broke the C ABI). Point the Dockerfile/CI at requirements.lock.txt, which
# pins a coherent numpy-1.x generation. (Those files are owned elsewhere — this
# is a flag, not a fix.)
#
# DEPENDENCY-CONFUSION FLAG: the Dockerfile/CI install these with
#   --extra-index-url https://download.pytorch.org/whl/cpu
# An *extra* index lets pip prefer a higher version from public PyPI over the
# trusted CPU-wheel host — the classic confusion shape. Prefer a single
# --index-url or exact/hashed pins. Flagged, not changed (Dockerfile/CI owned
# elsewhere).

# --- CV pipeline (pipeline/) ---
opencv-python-headless>=4.9.0
# numpy<2.0 is the authoritative constraint; requirements.lock.txt conforms to
# it (pins 1.26.4). Do NOT lift the ceiling without re-validating the whole CV
# stack against numpy 2.x's ABI break — easyocr in particular.
numpy>=1.26.0,<2.0
Pillow>=10.0.0
# HEIC/HEIF and AVIF decoding for pipeline/normalize_image.py. Neither OpenCV
# nor Pillow reads an iPhone HEIC without it; the manylinux wheels bundle
# libheif, so no apt package is needed in the Docker image. 1.x wheels are
# manylinux_2_28 (glibc 2.28+, which python:3.11-slim has) and need
# Pillow>=11.1, which the floor above allows.
pillow-heif>=1.0,<2
torch>=2.1.0
torchvision>=0.16.0
# Floor raised above the compromised ultralytics releases: 8.3.41 and 8.3.42
# (Dec 2024) shipped a cryptominer on PyPI and were later yanked. The old
# >=8.1.0 floor allowed them; >=8.3.43 cannot resolve to either.
ultralytics>=8.3.43
# Still required by pipeline/ocr_labels.py and pipeline/side_labels.py.
# pipeline/ocr_devices.py no longer uses it — see switch-ocr below.
easyocr>=1.7.1

# --- Device-label OCR (pipeline/ocr_devices.py) ---
# The switch_ocr package is vendored in this repo; these are its engines.
# Without them ocr_devices.py raises on import rather than falling back:
# a silent fall back to a weaker engine produces a confident WRONG inventory
# (EasyOCR read this rack's D-Link as "Calb"), which is worse than a scan that
# fails and can be retried. rapidocr is the lighter of the two backends and is
# what switch-ocr selects by default.
rapidocr>=2.0.0
onnxruntime>=1.17

# --- Vendor matrix / firmware scraper ---
beautifulsoup4>=4.12.0
openpyxl>=3.1.2
requests>=2.31.0
urllib3>=2.0.0

# --- Notification channels (Slack / Teams / Outlook) ---
msal>=1.30.0

# --- Netdisco bridge ---
psycopg2-binary>=2.9.9

# --- Shared utilities ---
python-dotenv>=1.0.0

# --- Firmware version lookup (firmware_lookup/, POST /api/firmware) ---
# Reads each vendor's own real site for the latest official firmware. requests
# is already pinned above. cryptography encrypts saved vendor logins at rest;
# pymupdf (imported as fitz) extracts version text from vendor PDF datasheets;
# playwright drives a real browser for the ~31 vendors behind a login or a
# JS-rendered page. There is NO fallback engine — a missing dependency makes
# firmware lookup fail loudly rather than return a wrong version.
#
# After installing these, run ONCE on the box that serves firmware lookups:
#     playwright install chromium firefox
# (chromium covers most vendors; firefox is required only for Lenovo.)
cryptography>=41.0
pymupdf>=1.23
playwright>=1.62.0
